project(PatricianSGL LANGUAGES CXX)

if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 4)

set(manifest "${CMAKE_CURRENT_SOURCE_DIR}/research/sgl-original/dispatch.json")

file(READ "${manifest}" dispatch)
string(JSON count LENGTH "${dispatch}" exports)
if(NOT count EQUAL 180)
    message(FATAL_ERROR "The verified SGL dispatch table must contain 180 exports.")
endif()

set(source "#include \"unimplemented.hpp\"\n")

math(EXPR last "${count} - 1")
foreach(index RANGE ${last})
    string(JSON name GET "${dispatch}" exports ${index} name)
    if(NOT name MATCHES "^sgl_[A-Za-z0-9_]+$" OR name IN_LIST seen)
        message(FATAL_ERROR "Invalid or duplicate SGL export: ${name}")
    endif()
    list(APPEND seen "${name}")
    string(JSON classification GET "${dispatch}" exports ${index} classification)
    if(classification STREQUAL "no_op")
        string(JSON instructions GET "${dispatch}" exports ${index} instructions)
        if(instructions STREQUAL "ret")
            string(APPEND source "extern \"C\" __declspec(naked) void __cdecl ${name}() { __asm { _emit 0xc3 } }\n")
        elseif(instructions STREQUAL "xor eax,eax; ret")
            string(APPEND source "extern \"C\" __declspec(naked) void __cdecl ${name}() { __asm { xor eax, eax } __asm { ret } }\n")
        else()
            message(FATAL_ERROR "Unknown no-op contract for ${name}: ${instructions}")
        endif()
    elseif(NOT name IN_LIST implemented_exports)
        string(APPEND source "extern \"C\" void __cdecl ${name}() { sgl::unimplemented(\"${name}\"); }\n")
    endif()
    string(APPEND definition "    ${name}\n")
endforeach()
foreach(name IN LISTS implemented_exports)
    if(NOT name IN_LIST seen)
        message(FATAL_ERROR "Implemented export is absent from the verified SGL dispatch table: ${name}")
    endif()
endforeach()

set(generated_source "${CMAKE_CURRENT_BINARY_DIR}/unimplemented_exports.cpp")
set(generated_definition "${CMAKE_CURRENT_BINARY_DIR}/exports.def")
file(WRITE "${generated_source}" "${source}")
file(WRITE "${generated_definition}" "${definition}")
